package app.eva.vpn

import android.util.Base64
import org.json.JSONArray
import org.json.JSONObject
import java.net.URI
import java.net.URLDecoder

data class Server(val name: String, val host: String, val port: Int, val out: JSONObject, val ping: Int = -1, val udp: Boolean = false)

object Parser {
    private fun b64(s: String) = String(Base64.decode(s.trim().replace('-', '+').replace('_', '/'), Base64.DEFAULT))
    private fun dec(s: String) = URLDecoder.decode(s, "UTF-8")

    fun parse(l: String): Server? = try {
        val s = l.trim()
        when {
            s.startsWith("vmess://") -> vmess(s)
            s.startsWith("ss://") -> ss(s)
            s.startsWith("hysteria2://") || s.startsWith("hy2://") -> hy2(s)
            s.startsWith("vless://") || s.startsWith("trojan://") -> uri(s)
            else -> null
        }
    } catch (e: Exception) { null }

    fun diag(body: String): String {
        val t = body.trim()
        val text = (if (t.contains("://")) t else try { b64(t) } catch (e: Exception) { t }).trim()
        val sch = Regex("(?m)^([a-zA-Z0-9]+)://").findAll(text).groupingBy { it.groupValues[1] }.eachCount()
            .entries.joinToString { "${it.key}×${it.value}" }
        return "ответ ${t.length} симв., ссылки: ${sch.ifEmpty { "нет" }}, начало: ${text.take(8)}"
    }

    fun links(body: String): List<Server> {
        val t = body.trim()
        val text = if (t.contains("://")) t else try { b64(t) } catch (e: Exception) { t }
        return text.lines().mapNotNull { parse(it) }
    }

    private fun base(type: String, host: String, port: Int) =
        JSONObject().put("type", type).put("tag", "proxy").put("server", host).put("server_port", port)

    private fun uri(link: String): Server {
        val frag = link.substringAfter('#', "")
        val u = URI(link.substringBefore('#').replace(" ", "%20"))
        val p = (u.rawQuery ?: "").split("&").filter { it.contains("=") }
            .associate { it.substringBefore("=") to dec(it.substringAfter("=")) }
        val host = u.host.trim('[', ']')
        val o = base(u.scheme, host, u.port)
        if (u.scheme == "vless") {
            o.put("uuid", u.userInfo)
            p["flow"]?.takeIf { it.isNotEmpty() }?.let { o.put("flow", it) }
        } else o.put("password", dec(u.userInfo))
        tls(o, p); transport(o, p)
        return Server(dec(frag.replace("+", "%2B")).ifEmpty { host }, host, u.port, o)
    }

    private fun tls(o: JSONObject, p: Map<String, String>) {
        val sec = p["security"] ?: if (o.getString("type") == "trojan") "tls" else "none"
        if (sec == "none") return
        val t = JSONObject().put("enabled", true)
        (p["sni"] ?: p["host"])?.takeIf { it.isNotEmpty() }?.let { t.put("server_name", it) }
        if (p["allowInsecure"] == "1") t.put("insecure", true)
        p["alpn"]?.takeIf { it.isNotEmpty() }?.let { t.put("alpn", JSONArray(it.split(","))) }
        t.put("utls", JSONObject().put("enabled", true).put("fingerprint", p["fp"] ?: "chrome"))
        if (sec == "reality") t.put("reality", JSONObject().put("enabled", true)
            .put("public_key", p["pbk"] ?: "").put("short_id", p["sid"] ?: ""))
        o.put("tls", t)
    }

    private fun transport(o: JSONObject, p: Map<String, String>) {
        val host = p["host"]?.takeIf { it.isNotEmpty() } ?: o.getString("server")
        when (p["type"]) {
            "ws" -> o.put("transport", JSONObject().put("type", "ws").put("path", p["path"] ?: "/")
                .put("headers", JSONObject().put("Host", host)))
            "httpupgrade" -> o.put("transport", JSONObject().put("type", "httpupgrade")
                .put("path", p["path"] ?: "/").put("host", host))
            "grpc" -> o.put("transport", JSONObject().put("type", "grpc").put("service_name", p["serviceName"] ?: ""))
        }
    }

    private fun vmess(link: String): Server {
        val j = JSONObject(b64(link.removePrefix("vmess://")))
        val host = j.getString("add"); val port = j.get("port").toString().toInt()
        val o = base("vmess", host, port).put("uuid", j.getString("id"))
            .put("security", j.optString("scy").ifEmpty { "auto" })
            .put("alter_id", j.optString("aid", "0").toIntOrNull() ?: 0)
        val p = mapOf(
            "type" to j.optString("net"), "host" to j.optString("host"), "path" to j.optString("path"),
            "security" to if (j.optString("tls") == "tls") "tls" else "none",
            "sni" to j.optString("sni"), "fp" to j.optString("fp")
        ).filterValues { it.isNotEmpty() }
        tls(o, p); transport(o, p)
        return Server(j.optString("ps", host), host, port, o)
    }

    private val hy = Regex("^(?:hysteria2|hy2)://([^@]+)@(\\[[^\\]]+\\]|[^:/?#]+):(\\d+)[,\\-\\d]*/?\\??([^#]*)#?(.*)$")

    private fun hy2(link: String): Server {
        val (pw, h, pt, qs, fr) = hy.matchEntire(link)!!.destructured
        val host = h.trim('[', ']'); val port = pt.toInt()
        val p = qs.split("&").filter { it.contains("=") }.associate { it.substringBefore("=") to dec(it.substringAfter("=")) }
        val t = JSONObject().put("enabled", true)
        (p["sni"] ?: p["peer"])?.takeIf { it.isNotEmpty() }?.let { t.put("server_name", it) }
        if (p["insecure"] == "1") t.put("insecure", true)
        p["alpn"]?.takeIf { it.isNotEmpty() }?.let { t.put("alpn", JSONArray(it.split(","))) }
        val o = base("hysteria2", host, port).put("password", dec(pw)).put("tls", t)
        p["obfs"]?.takeIf { it.isNotEmpty() }?.let {
            o.put("obfs", JSONObject().put("type", it).put("password", p["obfs-password"] ?: ""))
        }
        return Server(dec(fr).ifEmpty { host }, host, port, o, udp = true)
    }

    private fun ss(link: String): Server {
        val rest = link.removePrefix("ss://")
        val name = dec(rest.substringAfter('#', ""))
        var main = rest.substringBefore('#').substringBefore('?')
        if (!main.contains('@')) main = b64(main)
        val cred = main.substringBeforeLast('@'); val hp = main.substringAfterLast('@')
        val mp = if (cred.contains(':')) dec(cred) else b64(cred)
        val host = hp.substringBeforeLast(':').trim('[', ']'); val port = hp.substringAfterLast(':').toInt()
        val o = base("shadowsocks", host, port).put("method", mp.substringBefore(':')).put("password", mp.substringAfter(':'))
        return Server(name.ifEmpty { host }, host, port, o)
    }
}

package app.eva.vpn

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.net.HttpURLConnection
import java.net.InetSocketAddress
import java.net.Socket
import java.net.URL

data class Info(val up: Long, val down: Long, val total: Long, val expire: Long) {
    companion object {
        fun of(h: String): Info? {
            if (h.isBlank()) return null
            val m = h.split(";").mapNotNull {
                val p = it.trim().split("=")
                if (p.size == 2) p[0] to (p[1].toDoubleOrNull()?.toLong() ?: 0L) else null
            }.toMap()
            return Info(m["upload"] ?: 0L, m["download"] ?: 0L, m["total"] ?: 0L, m["expire"] ?: 0L)
        }
    }
}

object Sub {
    fun parse(body: String) = Parser.links(body)

    suspend fun load(url: String, hwid: String): Pair<String, String> = withContext(Dispatchers.IO) {
        val c = URL(url.trim()).openConnection() as HttpURLConnection
        c.setRequestProperty("User-Agent", "Happ/2.7.0")
        c.setRequestProperty("x-hwid", hwid)
        c.setRequestProperty("x-device-os", "Android")
        c.setRequestProperty("x-ver-os", android.os.Build.VERSION.RELEASE)
        c.setRequestProperty("x-device-model", android.os.Build.MODEL)
        c.connectTimeout = 15000; c.readTimeout = 15000
        val body = c.inputStream.bufferedReader().readText()
        body to (c.getHeaderField("subscription-userinfo") ?: "")
    }

    suspend fun ping(s: Server): Int = withContext(Dispatchers.IO) {
        fun tcp(port: Int): Int = try {
            val t = System.nanoTime()
            Socket().use { it.connect(InetSocketAddress(s.host, port), 3000) }
            maxOf(1, ((System.nanoTime() - t) / 1_000_000).toInt())
        } catch (e: Exception) { -1 }
        val r = tcp(s.port)
        if (r > 0) r else if (s.udp) { val r2 = tcp(443); if (r2 > 0) r2 else -2 } else 9999
    }
}
