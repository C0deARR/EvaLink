package app.eva.vpn

import android.content.SharedPreferences
import org.json.JSONArray
import org.json.JSONObject
import java.net.Inet4Address
import java.net.InetAddress

object Cfg {
    // адрес сервера превращаем в IP заранее: у ядра на Android нет своего DNS для этого
    private fun resolve(h: String): String = try {
        InetAddress.getAllByName(h).firstOrNull { it is Inet4Address }?.hostAddress ?: h
    } catch (e: Exception) { h }

    fun build(proxy: JSONObject, p: SharedPreferences): JSONObject {
        val px = JSONObject(proxy.toString())
        val host = px.getString("server")
        val ip = resolve(host)
        if (ip != host) {
            px.put("server", ip)
            px.optJSONObject("tls")?.let { if (!it.has("server_name")) it.put("server_name", host) }
        }
        val dnsUrl = p.getString("dns", "https://1.1.1.1/dns-query") ?: "https://1.1.1.1/dns-query"
        val dns = JSONObject()
            .put("servers", JSONArray().put(JSONObject().put("tag", "r").put("address", dnsUrl).put("detour", "proxy")))
            .put("final", "r")
        if (p.getBoolean("v4", true)) dns.put("strategy", "ipv4_only")
        val rules = JSONArray()
            .put(JSONObject().put("action", "sniff"))
            .put(JSONObject().put("protocol", "dns").put("action", "hijack-dns"))
        if (p.getBoolean("lan", true)) rules.put(JSONObject().put("ip_is_private", true).put("outbound", "direct"))
        return JSONObject()
            .put("log", JSONObject().put("level", if (p.getBoolean("debug", false)) "debug" else "info"))
            .put("dns", dns)
            .put("inbounds", JSONArray().put(JSONObject().put("type", "socks").put("tag", "socks-in")
                .put("listen", "127.0.0.1").put("listen_port", 10808)).put(JSONObject().put("type", "tun").put("tag", "tun-in")
                .put("address", JSONArray().put("172.19.0.1/30")).put("mtu", p.getInt("mtu", 1500))
                .put("auto_route", false).put("stack", "mixed")))
            .put("outbounds", JSONArray().put(px).put(JSONObject().put("type", "direct").put("tag", "direct")))
            .put("route", JSONObject().put("rules", rules).put("final", "proxy"))
    }
}

class EvaVpn : VpnService(), PlatformInterface {
    companion object {
        const val ACTION_STATE = "app.eva.vpn.STATE"
        const val ACTION_QUERY = "app.eva.vpn.QUERY"
        const val CH = "eva_vpn"
        @Volatile var running = false
        @Volatile var error = ""
        private val logs = java.util.ArrayDeque<String>()

        fun addLog(s: String) { synchronized(logs) { logs.addLast(s); while (logs.size > 300) logs.removeFirst() } }
        fun publish(c: Context) {
            val text = synchronized(logs) { logs.toList().takeLast(80).joinToString("\n") }
            c.sendBroadcast(Intent(ACTION_STATE).setPackage(c.packageName)
                .putExtra("on", running).putExtra("err", error).putExtra("log", text))
        }
    }

    private var box: BoxService? = null
    private var apps: List<String> = emptyList()
    private var mode = "except"
    private var mtu = 1500
    private val h = Handler(Looper.getMainLooper())
    private var pubPending = false

    override fun onStartCommand(i: Intent?, f: Int, id: Int): Int {
        if (i?.action == "stop") { shutdown(); return START_NOT_STICKY }
        startForeground(1, notif(i?.getStringExtra("name") ?: ""))
        val cfg = i?.getStringExtra("cfg")
        if (i == null || cfg == null) { shutdown(); return START_NOT_STICKY }
        apps = i.getStringArrayListExtra("apps") ?: emptyList()
        mode = i.getStringExtra("mode") ?: "except"
        mtu = i.getIntExtra("mtu", 1500)
        Thread {
            try {
                stopBox()
                error = ""
                addLog("— запуск ядра —")
                Libbox.setup(SetupOptions().apply {
                    basePath = filesDir.path; workingPath = filesDir.path; tempPath = cacheDir.path
                })
                box = Libbox.newService(cfg, this).also { it.start() }
                running = true
                addLog("— ядро запущено —")
                publish(this)
            } catch (e: Throwable) {
                error = e.message ?: e.toString()
                addLog("ОШИБКА: $error")
                shutdown()
            }
        }.start()
        return START_NOT_STICKY
    }

    private fun notif(name: String): Notification {
        val nm = getSystemService(NotificationManager::class.java)
        if (Build.VERSION.SDK_INT >= 26) nm.createNotificationChannel(NotificationChannel(CH, "VPN", NotificationManager.IMPORTANCE_LOW))
        val stop = PendingIntent.getService(this, 0, Intent(this, EvaVpn::class.java).setAction("stop"),
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT)
        val open = PendingIntent.getActivity(this, 0, Intent(this, MainActivity::class.java), PendingIntent.FLAG_IMMUTABLE)
        val b = if (Build.VERSION.SDK_INT >= 26) Notification.Builder(this, CH) else Notification.Builder(this)
        return b.setSmallIcon(android.R.drawable.ic_lock_lock).setContentTitle("EvaLink")
            .setContentText(if (name.isEmpty()) "VPN активен" else name).setContentIntent(open).setOngoing(true)
            .addAction(Notification.Action.Builder(Icon.createWithResource(this, android.R.drawable.ic_delete), "Отключить", stop).build())
            .build()
    }

    private fun stopBox() {
        try { box?.close() } catch (_: Throwable) {}
        box = null
    }

    // Выключение: закрываем ядро и затем завершаем процесс :vpn.
    // Смерть процесса гарантированно закрывает туннель, даже если ядро зависло.
    private fun shutdown() {
        running = false
        publish(this)
        Thread {
            try { Thread.sleep(5000) } catch (_: InterruptedException) {}
            android.os.Process.killProcess(android.os.Process.myPid())
        }.start()
        Thread {
            stopBox()
            try { stopForeground(STOP_FOREGROUND_REMOVE) } catch (_: Throwable) {}
            publish(this)
            try { Thread.sleep(500) } catch (_: InterruptedException) {}
            android.os.Process.killProcess(android.os.Process.myPid())
        }.start()
    }

    override fun onDestroy() { running = false; super.onDestroy() }
    override fun onRevoke() { shutdown() }

    // Само приложение (в нём работает ядро) исключено из туннеля, иначе трафик ядра зациклится
    override fun openTun(o: TunOptions?): Int {
        val b = Builder().setSession("EvaLink").setMtu(mtu)
            .addAddress("172.19.0.1", 30).addRoute("0.0.0.0", 0).addDnsServer("172.19.0.2")
        val list = apps.filter { it != packageName }
        if (mode == "only" && list.isNotEmpty()) {
            for (p in list) try { b.addAllowedApplication(p) } catch (_: Exception) {}
        } else {
            try { b.addDisallowedApplication(packageName) } catch (_: Exception) {}
            for (p in list) try { b.addDisallowedApplication(p) } catch (_: Exception) {}
        }
        return b.establish()!!.detachFd()
    }

    override fun startDefaultInterfaceMonitor(l: InterfaceUpdateListener?) {}
    override fun closeDefaultInterfaceMonitor(l: InterfaceUpdateListener?) {}

    override fun writeLog(message: String?) {
        if (message == null) return
        addLog(message)
        if (!pubPending) { pubPending = true; h.postDelayed({ pubPending = false; publish(this) }, 800) }
    }

    override fun autoDetectInterfaceControl(fd: Int) { protect(fd) }
    override fun useProcFS() = false
    override fun findConnectionOwner(a: Int, b: String?, c: Int, d: String?, e: Int): Int = 0

    override fun uidByPackageName(n: String?): Int = 0

}

package app.eva.vpn

import android.util.Base64
import org.json.JSONArray
import org.json.JSONObject
import java.net.HttpURLConnection

// Админские операции: запись файлов в публичный репозиторий с новостями через GitHub API.
// Токен хранится только на телефоне админа и в проект не попадает.
object Gh {
    // Публичный репозиторий для новостей и фото тестеров (создай его на GitHub, названия ниже должны совпадать)
    const val REPO = "C0deARR/EvaLink-news"

    private fun api(path: String) = "https://api.github.com/repos/$REPO/contents/$path"

    private fun call(url: String, method: String, token: String, body: String?): Pair<Int, String> {
        val c = java.net.URL(url).openConnection() as HttpURLConnection
        c.requestMethod = method
        c.setRequestProperty("Authorization", "Bearer $token")
        c.setRequestProperty("Accept", "application/vnd.github+json")
        c.setRequestProperty("User-Agent", "EvaLink")
        c.connectTimeout = 20000; c.readTimeout = 60000
        if (body != null) {
            c.doOutput = true
            c.setRequestProperty("Content-Type", "application/json")
            c.outputStream.use { it.write(body.toByteArray()) }
        }
        val code = c.responseCode
        val text = (if (code < 400) c.inputStream else c.errorStream)?.bufferedReader()?.readText() ?: ""
        return code to text
    }

    private fun msg(t: String) = try { JSONObject(t).optString("message") } catch (e: Exception) { t.take(80) }

    // (sha, текст) или null, если файла нет
    fun get(token: String, path: String): Pair<String, String>? {
        val (code, t) = call(api(path), "GET", token, null)
        if (code == 404) return null
        if (code >= 400) throw Exception("GitHub $code: ${msg(t)}")
        val j = JSONObject(t)
        return j.getString("sha") to String(Base64.decode(j.getString("content"), Base64.DEFAULT), Charsets.UTF_8)
    }

    fun put(token: String, path: String, bytes: ByteArray, message: String, sha: String?) {
        val j = JSONObject().put("message", message).put("content", Base64.encodeToString(bytes, Base64.NO_WRAP))
        if (sha != null) j.put("sha", sha)
        val (code, t) = call(api(path), "PUT", token, j.toString())
        if (code >= 400) throw Exception("GitHub $code: ${msg(t)}")
    }

    fun sendNews(token: String, title: String, text: String, url: String) {
        val j = JSONObject().put("id", System.currentTimeMillis() / 1000).put("title", title).put("text", text).put("url", url)
        put(token, "news.json", j.toString(2).toByteArray(), "news", get(token, "news.json")?.first)
    }

// фото из галереи: уменьшаем, поворачиваем по EXIF и пережимаем в JPEG
private fun prepJpeg(ctx: Context, uri: Uri): ByteArray? = try {
    val cr = ctx.contentResolver
    val o = BitmapFactory.Options().apply { inJustDecodeBounds = true }
    cr.openInputStream(uri)?.use { BitmapFactory.decodeStream(it, null, o) }
    var s = 1
    while (o.outWidth / s > 1600 || o.outHeight / s > 1600) s *= 2
    var bmp = cr.openInputStream(uri)?.use { BitmapFactory.decodeStream(it, null, BitmapFactory.Options().apply { inSampleSize = s }) }
    val rot = try { cr.openInputStream(uri)?.use { android.media.ExifInterface(it).getAttributeInt(android.media.ExifInterface.TAG_ORIENTATION, 1) } ?: 1 } catch (e: Exception) { 1 }
    val deg = when (rot) { 6 -> 90f; 3 -> 180f; 8 -> 270f; else -> 0f }
    if (bmp != null && deg != 0f) bmp = Bitmap.createBitmap(bmp, 0, 0, bmp.width, bmp.height, Matrix().apply { postRotate(deg) }, true)
    bmp?.let { val bo = java.io.ByteArrayOutputStream(); it.compress(Bitmap.CompressFormat.JPEG, 85, bo); bo.toByteArray() }
} catch (e: Exception) { null }

object News {
    // news.json лежит в публичном репозитории Gh.REPO; админ обновляет его из приложения
    const val NEWS_URL = "https://raw.githubusercontent.com/${Gh.REPO}/main/news.json"

    // проверка раз в ~6 часов, пока приложение установлено
    fun schedule(c: Context) {
        val am = c.getSystemService(AlarmManager::class.java)
        val pi = PendingIntent.getBroadcast(c, 3, Intent(c, NewsReceiver::class.java),
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT)
        am.setInexactRepeating(AlarmManager.RTC, System.currentTimeMillis() + 3600000L, 6 * 3600000L, pi)
    }

    // true, если появилось новое объявление (id больше прежнего)
    fun check(c: Context, notify: Boolean): Boolean {
        val con = java.net.URL("$NEWS_URL?t=${System.currentTimeMillis()}").openConnection() as HttpURLConnection
        con.connectTimeout = 10000; con.readTimeout = 10000
        val j = JSONObject(con.inputStream.bufferedReader().readText())
        val id = j.optLong("id", 0L)
        val p = c.getSharedPreferences("eva", Context.MODE_PRIVATE)
        if (id <= 0L || id <= p.getLong("news_id", 0L)) return false
        val title = j.optString("title"); val text = j.optString("text")
        p.edit().putLong("news_id", id).putString("news_title", title).putString("news_text", text)
            .putString("news_url", j.optString("url")).apply()
        if (notify && p.getBoolean("news", true)) post(c, title, text)
        return true
    }

    private fun post(c: Context, title: String, text: String) {
        val nm = c.getSystemService(NotificationManager::class.java)
        if (Build.VERSION.SDK_INT >= 26) nm.createNotificationChannel(NotificationChannel("eva_news", "Новости", NotificationManager.IMPORTANCE_DEFAULT))
        val open = PendingIntent.getActivity(c, 0, Intent(c, MainActivity::class.java), PendingIntent.FLAG_IMMUTABLE)
        val b = if (Build.VERSION.SDK_INT >= 26) Notification.Builder(c, "eva_news") else Notification.Builder(c)
        nm.notify(3, b.setSmallIcon(android.R.drawable.ic_dialog_info).setContentTitle(title.ifEmpty { "EvaLink" })
            .setContentText(text).setStyle(Notification.BigTextStyle().bigText(text))
            .setContentIntent(open).setAutoCancel(true).build())
    }

package app.eva.vpn

import android.util.Base64
import org.json.JSONArray
import org.json.JSONObject
import java.net.URI
import java.net.URLDecoder

data class Server(val name: String, val host: String, val port: Int, val out: JSONObject, val ping: Int = -1, val udp: Boolean = false, val bad: String = "")

object Parser {
    private val skip = mutableListOf<String>()
    fun skipReport() = if (skip.isEmpty()) "" else ", пропущено: " + skip.groupingBy { it }.eachCount().entries.joinToString { "${it.key}×${it.value}" }
    private fun b64(s: String) = String(Base64.decode(s.trim().replace('-', '+').replace('_', '/'), Base64.DEFAULT))
    private fun dec(s: String) = URLDecoder.decode(s, "UTF-8")

    fun parse(l: String): Server? = try {
        val s = l.trim()
        when {
            s.startsWith("vmess://") -> vmess(s)
            s.startsWith("ss://") -> ss(s)
            s.startsWith("hysteria2://") || s.startsWith("hy2://") -> hy2(s)
            s.startsWith("vless://") || s.startsWith("trojan://") -> uri(s)
            else -> null
        }
    } catch (e: Exception) { null }

    fun diag(body: String): String {
        val t = body.trim()
        val text = (if (t.contains("://")) t else try { b64(t) } catch (e: Exception) { t }).trim()
        val sch = Regex("(?m)^([a-zA-Z0-9]+)://").findAll(text).groupingBy { it.groupValues[1] }.eachCount()
            .entries.joinToString { "${it.key}×${it.value}" }
        return "ответ ${t.length} симв., ссылки: ${sch.ifEmpty { "нет" }}, начало: ${text.take(8)}"
    }

    private fun xray(c: JSONObject): Server? {
        val outs = c.optJSONArray("outbounds") ?: return null
        val list = (0 until outs.length()).map { outs.getJSONObject(it) }
        val known = setOf("vless", "vmess", "trojan", "shadowsocks", "hysteria")
        val ob = list.firstOrNull { it.optString("tag") == "proxy" && it.optString("protocol") in known }
            ?: list.firstOrNull { it.optString("protocol") in known }
            ?: run { skip += list.map { it.optString("protocol") }.filter { it.isNotEmpty() }.distinct().joinToString("/").ifEmpty { "нет outbound" }; return null }
        val proto = ob.optString("protocol")
        val st = ob.optJSONObject("streamSettings") ?: JSONObject()
        val s = ob.optJSONObject("settings") ?: return null
        val host: String; val port: Int; val o: JSONObject; var udp = false
        when (proto) {
            "vless", "vmess" -> {
                val v = s.getJSONArray("vnext").getJSONObject(0)
                val u = v.getJSONArray("users").getJSONObject(0)
                host = v.getString("address"); port = v.getInt("port")
                o = base(proto, host, port).put("uuid", u.getString("id"))
                if (proto == "vless") u.optString("flow").takeIf { it.isNotEmpty() }?.let { o.put("flow", it) }
                else o.put("security", u.optString("security").ifEmpty { "auto" }).put("alter_id", u.optInt("alterId", 0))
            }
            "trojan", "shadowsocks" -> {
                val v = s.getJSONArray("servers").getJSONObject(0)
                host = v.getString("address"); port = v.getInt("port")
                o = base(proto, host, port).put("password", v.getString("password"))
                if (proto == "shadowsocks") o.put("method", v.getString("method"))
            }
            "hysteria" -> {
                host = s.getString("address"); port = s.getInt("port"); udp = true
                o = base("hysteria2", host, port).put("password", st.optJSONObject("hysteriaSettings")?.optString("auth") ?: "")
            }
            else -> return null
        }
        val net = st.optString("network", "tcp"); val sec = st.optString("security", "none")
        if (sec == "tls" || sec == "reality") {
            val t = JSONObject().put("enabled", true)
            val ts = st.optJSONObject(if (sec == "tls") "tlsSettings" else "realitySettings") ?: JSONObject()
            ts.optString("serverName").takeIf { it.isNotEmpty() }?.let { t.put("server_name", it) }
            if (ts.optBoolean("allowInsecure")) t.put("insecure", true)
            ts.optJSONArray("alpn")?.let { t.put("alpn", it) }
            if (proto != "hysteria") t.put("utls", JSONObject().put("enabled", true).put("fingerprint", ts.optString("fingerprint").ifEmpty { "chrome" }))
            if (sec == "reality") t.put("reality", JSONObject().put("enabled", true)
                .put("public_key", ts.optString("publicKey")).put("short_id", ts.optString("shortId")))
            o.put("tls", t)
        }
        var bad = ""
        when (net) {
            "tcp", "raw", "hysteria" -> {}
            "ws" -> {
                val w = st.optJSONObject("wsSettings") ?: JSONObject()
                val h = w.optJSONObject("headers")?.optString("Host")?.takeIf { it.isNotEmpty() }
                    ?: w.optString("host").ifEmpty { host }
                o.put("transport", JSONObject().put("type", "ws").put("path", w.optString("path").ifEmpty { "/" })
                    .put("headers", JSONObject().put("Host", h)))
            }
            "grpc" -> o.put("transport", JSONObject().put("type", "grpc")
                .put("service_name", st.optJSONObject("grpcSettings")?.optString("serviceName") ?: ""))
            "httpupgrade" -> {
                val w = st.optJSONObject("httpupgradeSettings") ?: JSONObject()
                o.put("transport", JSONObject().put("type", "httpupgrade").put("path", w.optString("path").ifEmpty { "/" })
                    .put("host", w.optString("host").ifEmpty { host }))
            }
            else -> bad = net.uppercase()
        }
        return Server(c.optString("remarks").ifEmpty { host }, host, port, o, udp = udp, bad = bad)
    }

    private fun jsonServers(t: String): List<Server> =
        if (t.startsWith("[")) JSONArray(t).let { a -> (0 until a.length()).mapNotNull { i -> try { xray(a.getJSONObject(i)) } catch (e: Exception) { skip += "ошибка разбора"; null } } }
        else listOfNotNull(xray(JSONObject(t)))

    fun links(body: String): List<Server> {
        val t = body.trim()
        skip.clear()
        if (t.startsWith("[") || t.startsWith("{")) return try { jsonServers(t) } catch (e: Exception) { emptyList() }
        val text = if (t.contains("://")) t else try { b64(t) } catch (e: Exception) { t }
        return text.lines().mapNotNull { parse(it) }
    }

    private fun base(type: String, host: String, port: Int): JSONObject {
        val o = JSONObject().put("type", type).put("tag", "proxy").put("server", host).put("server_port", port)
        if (type == "vless") o.put("packet_encoding", "xudp")
        return o
    }

    private fun uri(link: String): Server {
        val frag = link.substringAfter('#', "")
        val u = URI(link.substringBefore('#').replace(" ", "%20"))
        val p = (u.rawQuery ?: "").split("&").filter { it.contains("=") }
            .associate { it.substringBefore("=") to dec(it.substringAfter("=")) }
        val host = u.host.trim('[', ']')
        val o = base(u.scheme, host, u.port)
        if (u.scheme == "vless") {
            o.put("uuid", u.userInfo)
            p["flow"]?.takeIf { it.isNotEmpty() }?.let { o.put("flow", it) }
        } else o.put("password", dec(u.userInfo))
        tls(o, p); transport(o, p)
        return Server(dec(frag.replace("+", "%2B")).ifEmpty { host }, host, u.port, o)
    }

    private fun tls(o: JSONObject, p: Map<String, String>) {
        val sec = p["security"] ?: if (o.getString("type") == "trojan") "tls" else "none"
        if (sec == "none") return
        val t = JSONObject().put("enabled", true)
        (p["sni"] ?: p["host"])?.takeIf { it.isNotEmpty() }?.let { t.put("server_name", it) }
        if (p["allowInsecure"] == "1") t.put("insecure", true)
        p["alpn"]?.takeIf { it.isNotEmpty() }?.let { t.put("alpn", JSONArray(it.split(","))) }
        t.put("utls", JSONObject().put("enabled", true).put("fingerprint", p["fp"] ?: "chrome"))
        if (sec == "reality") t.put("reality", JSONObject().put("enabled", true)
            .put("public_key", p["pbk"] ?: "").put("short_id", p["sid"] ?: ""))
        o.put("tls", t)
    }

    private fun transport(o: JSONObject, p: Map<String, String>) {
        val host = p["host"]?.takeIf { it.isNotEmpty() } ?: o.getString("server")
        when (p["type"]) {
            "ws" -> o.put("transport", JSONObject().put("type", "ws").put("path", p["path"] ?: "/")
                .put("headers", JSONObject().put("Host", host)))
            "httpupgrade" -> o.put("transport", JSONObject().put("type", "httpupgrade")
                .put("path", p["path"] ?: "/").put("host", host))
            "grpc" -> o.put("transport", JSONObject().put("type", "grpc").put("service_name", p["serviceName"] ?: ""))
        }
    }

    private fun vmess(link: String): Server {
        val j = JSONObject(b64(link.removePrefix("vmess://")))
        val host = j.getString("add"); val port = j.get("port").toString().toInt()
        val o = base("vmess", host, port).put("uuid", j.getString("id"))
            .put("security", j.optString("scy").ifEmpty { "auto" })
            .put("alter_id", j.optString("aid", "0").toIntOrNull() ?: 0)
        val p = mapOf(
            "type" to j.optString("net"), "host" to j.optString("host"), "path" to j.optString("path"),
            "security" to if (j.optString("tls") == "tls") "tls" else "none",
            "sni" to j.optString("sni"), "fp" to j.optString("fp")
        ).filterValues { it.isNotEmpty() }
        tls(o, p); transport(o, p)
        return Server(j.optString("ps", host), host, port, o)
    }

    private val hy = Regex("^(?:hysteria2|hy2)://([^@]+)@(\\[[^\\]]+\\]|[^:/?#]+):(\\d+)[,\\-\\d]*/?\\??([^#]*)#?(.*)$")

    private fun hy2(link: String): Server {
        val (pw, h, pt, qs, fr) = hy.matchEntire(link)!!.destructured
        val host = h.trim('[', ']'); val port = pt.toInt()
        val p = qs.split("&").filter { it.contains("=") }.associate { it.substringBefore("=") to dec(it.substringAfter("=")) }
        val t = JSONObject().put("enabled", true)
        (p["sni"] ?: p["peer"])?.takeIf { it.isNotEmpty() }?.let { t.put("server_name", it) }
        if (p["insecure"] == "1") t.put("insecure", true)
        p["alpn"]?.takeIf { it.isNotEmpty() }?.let { t.put("alpn", JSONArray(it.split(","))) }
        val o = base("hysteria2", host, port).put("password", dec(pw)).put("tls", t)
        p["obfs"]?.takeIf { it.isNotEmpty() }?.let {
            o.put("obfs", JSONObject().put("type", it).put("password", p["obfs-password"] ?: ""))
        }
        return Server(dec(fr).ifEmpty { host }, host, port, o, udp = true)
    }

    private fun ss(link: String): Server {
        val rest = link.removePrefix("ss://")
        val name = dec(rest.substringAfter('#', ""))
        var main = rest.substringBefore('#').substringBefore('?')
        if (!main.contains('@')) main = b64(main)
        val cred = main.substringBeforeLast('@'); val hp = main.substringAfterLast('@')
        val mp = if (cred.contains(':')) dec(cred) else b64(cred)
        val host = hp.substringBeforeLast(':').trim('[', ']'); val port = hp.substringAfterLast(':').toInt()
        val o = base("shadowsocks", host, port).put("method", mp.substringBefore(':')).put("password", mp.substringAfter(':'))
        return Server(name.ifEmpty { host }, host, port, o)
    }
}

package app.eva.vpn

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import org.json.JSONObject
import java.net.HttpURLConnection
import java.net.InetSocketAddress
import java.net.Socket
import java.net.URL

data class Info(val up: Long, val down: Long, val total: Long, val expire: Long) {
    companion object {
        fun of(h: String): Info? {
            if (h.isBlank()) return null
            val m = h.split(";").mapNotNull {
                val p = it.trim().split("=")
                if (p.size == 2) p[0] to (p[1].toDoubleOrNull()?.toLong() ?: 0L) else null
            }.toMap()
            return Info(m["upload"] ?: 0L, m["download"] ?: 0L, m["total"] ?: 0L, m["expire"] ?: 0L)
        }
    }
}

object Sub {
    fun parse(body: String) = Parser.links(body)

    suspend fun load(url: String, hwid: String): Pair<String, String> = withContext(Dispatchers.IO) {
        val c = URL(url.trim()).openConnection() as HttpURLConnection
        c.setRequestProperty("User-Agent", "Happ/2.7.0")
        c.setRequestProperty("x-hwid", hwid)
        c.setRequestProperty("x-device-os", "Android")
        c.setRequestProperty("x-ver-os", android.os.Build.VERSION.RELEASE)
        c.setRequestProperty("x-device-model", android.os.Build.MODEL)
        c.connectTimeout = 15000; c.readTimeout = 15000
        val body = c.inputStream.bufferedReader().readText()
        body to (c.getHeaderField("subscription-userinfo") ?: "")
    }

    suspend fun myIp(): String = withContext(Dispatchers.IO) {
        // запрос идёт через локальный SOCKS ядра: показывает настоящий IP сервера и проверяет, что прокси живой
        val px = java.net.Proxy(java.net.Proxy.Type.SOCKS, InetSocketAddress("127.0.0.1", 10808))
        val c = URL("https://ipinfo.io/json").openConnection(px) as HttpURLConnection
        c.connectTimeout = 8000; c.readTimeout = 8000
        val j = JSONObject(c.inputStream.bufferedReader().readText())
        "${j.optString("ip")} · ${j.optString("country")} · ${j.optString("org").take(26)}"
    }

    suspend fun ping(s: Server): Int = withContext(Dispatchers.IO) {
        fun tcp(port: Int): Int = try {
            val t = System.nanoTime()
            Socket().use { it.connect(InetSocketAddress(s.host, port), 3000) }
            maxOf(1, ((System.nanoTime() - t) / 1_000_000).toInt())
        } catch (e: Exception) { -1 }
        val r = tcp(s.port)
        if (r > 0) r else if (s.udp) { val r2 = tcp(443); if (r2 > 0) r2 else -2 } else 9999
    }
}
